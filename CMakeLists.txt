cmake_minimum_required(VERSION 3.25)
project(OrcaFLTK VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
list(PREPEND CMAKE_PREFIX_PATH "/opt/fltk-vr/toolkit")
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTS REQUIRED IMPORTED_TARGET fontconfig freetype2 zlib)
# Use the private SDK headers with the existing system shared libcurl runtime.
find_path(HTTP_CURL_INCLUDE curl/curl.h HINTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sdk/usr/include/x86_64-linux-gnu" REQUIRED)
find_library(HTTP_CURL_LIBRARY NAMES curl libcurl.so.4 HINTS /usr/lib/x86_64-linux-gnu REQUIRED)
add_library(slicer-transfer STATIC src/transfer.cpp)
target_include_directories(slicer-transfer PUBLIC src PRIVATE "${HTTP_CURL_INCLUDE}")
target_link_libraries(slicer-transfer PUBLIC Threads::Threads "${HTTP_CURL_LIBRARY}")
target_compile_options(slicer-transfer PRIVATE -Wall -Wextra -Wpedantic)
set(ORCA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../orca-upstream" CACHE PATH "Pinned Orca source checkout")
add_library(slicer-data STATIC src/profiles.cpp src/geometry.cpp)
target_include_directories(slicer-data PUBLIC src)
target_compile_options(slicer-data PRIVATE -Wall -Wextra -Wpedantic)
add_executable(orca-fltk src/main.cpp src/job.cpp src/viewport.cpp)
target_link_libraries(orca-fltk PRIVATE slicer-data slicer-transfer fltk::fltk-shared fltk::images-shared fltk::gl-shared OpenGL::GL Threads::Threads PkgConfig::FONTS)
target_include_directories(orca-fltk PRIVATE third_party/domed)
target_compile_definitions(orca-fltk PRIVATE DEFAULT_ORCA_RESOURCES="${ORCA_SOURCE_DIR}/resources" DEFAULT_ORCA_WORKER="${CMAKE_CURRENT_SOURCE_DIR}/build-engine/orca-fltk-worker")
set_target_properties(orca-fltk PROPERTIES INSTALL_RPATH "/opt/fltk-vr/bdf-runtime/lib;/opt/fltk-vr/toolkit/lib")
include(GNUInstallDirs)
install(TARGETS orca-fltk RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/orca-fltk COMPONENT OrcaRuntime)
install(PROGRAMS packaging/orca-fltk DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT OrcaRuntime)
install(FILES packaging/org.domed.OrcaSlicer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications COMPONENT OrcaRuntime)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/orca-fltk COMPONENT OrcaRuntime)
include(CTest)
if(BUILD_TESTING)
 add_executable(slicer-data-tests tests/data.cpp)
 target_link_libraries(slicer-data-tests PRIVATE slicer-data)
 target_compile_options(slicer-data-tests PRIVATE -UNDEBUG)
 add_test(NAME profiles-and-gcode COMMAND slicer-data-tests)
endif()
if(BUILD_TESTING)
 add_executable(slicer-job-tests tests/job.cpp src/job.cpp)
 target_link_libraries(slicer-job-tests PRIVATE slicer-data Threads::Threads)
 target_compile_options(slicer-job-tests PRIVATE -UNDEBUG)
 add_test(NAME worker-process-lifecycle COMMAND slicer-job-tests)
 set_tests_properties(worker-process-lifecycle PROPERTIES TIMEOUT 10)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build-engine/orca-fltk-worker")
 add_executable(slicer-ui-flow tests/ui-flow.cpp src/job.cpp src/viewport.cpp)
 target_link_libraries(slicer-ui-flow PRIVATE slicer-data slicer-transfer fltk::fltk-shared fltk::images-shared fltk::gl-shared OpenGL::GL Threads::Threads PkgConfig::FONTS)
 target_include_directories(slicer-ui-flow PRIVATE third_party/domed)
 target_compile_definitions(slicer-ui-flow PRIVATE DEFAULT_ORCA_RESOURCES="${ORCA_SOURCE_DIR}/resources" DEFAULT_ORCA_WORKER="${CMAKE_CURRENT_SOURCE_DIR}/build-engine/orca-fltk-worker")
 target_compile_options(slicer-ui-flow PRIVATE -UNDEBUG)
endif()
install(DIRECTORY "${ORCA_SOURCE_DIR}/resources/profiles" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/orca-fltk/resources COMPONENT OrcaRuntime)
install(FILES README.rst sources.json DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/orca-fltk COMPONENT OrcaRuntime)

if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(slicer-transfer-tests tests/transfer.cpp)
 target_link_libraries(slicer-transfer-tests PRIVATE slicer-transfer)
 add_test(NAME network-slice-transfer COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/transfer.py" $<TARGET_FILE:slicer-transfer-tests>)
 set_tests_properties(network-slice-transfer PROPERTIES TIMEOUT 40)
endif()
