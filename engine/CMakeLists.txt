# Build Orca's engine without its application, wxWidgets, WebKit or Python UI.
cmake_minimum_required(VERSION 3.25)
project(orca_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ORCA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../orca-upstream" CACHE PATH "Pinned OrcaSlicer checkout")
set(SLIC3R_STATIC OFF)
set(SLIC3R_GUI OFF)
set(SLIC3R_CAD OFF)
set(SLIC3R_PCH OFF)
set(SLIC3R_PROFILE OFF)
list(PREPEND CMAKE_MODULE_PATH "${ORCA_SOURCE_DIR}/cmake/modules")
include("${ORCA_SOURCE_DIR}/version.inc")
set(SLIC3R_BUILD_ID "${SoftFever_VERSION}+domed")
find_package(Threads REQUIRED)
find_package(Boost 1.83 CONFIG REQUIRED COMPONENTS filesystem thread log log_setup locale regex chrono atomic date_time iostreams program_options nowide)
add_library(boost_headeronly INTERFACE)
target_include_directories(boost_headeronly SYSTEM INTERFACE ${Boost_INCLUDE_DIRS})
add_library(boost_libs INTERFACE)
target_link_libraries(boost_libs INTERFACE boost_headeronly ${Boost_LIBRARIES})
add_compile_definitions(BOOST_LOG_DYN_LINK BOOST_ALL_NO_LIB)
find_package(Eigen3 5.0.1 CONFIG REQUIRED)
find_package(TBB CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(EXPAT REQUIRED)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(cereal CONFIG REQUIRED)
# Orca compiles its volume utilities into the core even for FFF jobs.
find_library(OPENVDB_LIBRARY NAMES openvdb REQUIRED)
find_library(IMATH_LIBRARY NAMES Imath-3_1 Imath REQUIRED)
add_library(OpenVDB::openvdb SHARED IMPORTED)
set_target_properties(OpenVDB::openvdb PROPERTIES IMPORTED_LOCATION "${OPENVDB_LIBRARY}" INTERFACE_LINK_LIBRARIES "${IMATH_LIBRARY}")
find_package(NLopt CONFIG REQUIRED)
if(NOT TARGET NLopt::nlopt AND TARGET NLopt::nlopt_cxx)
 add_library(NLopt::nlopt ALIAS NLopt::nlopt_cxx)
endif()
find_library(LIBNOISE_LIBRARY_RELEASE NAMES noise REQUIRED)
find_package(libnoise REQUIRED)
find_package(CURL REQUIRED)
add_library(libcurl INTERFACE)
target_link_libraries(libcurl INTERFACE CURL::libcurl OpenSSL::SSL OpenSSL::Crypto)
# Debian ships modular OpenCV libraries; Orca's deps package exposes opencv_world.
find_path(OpenCV_INCLUDE_DIR opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
find_library(OpenCV_CORE NAMES opencv_core REQUIRED)
find_library(OpenCV_IMGPROC NAMES opencv_imgproc REQUIRED)
find_library(OpenCV_IMGCODECS NAMES opencv_imgcodecs REQUIRED)
add_library(opencv_world INTERFACE)
target_include_directories(opencv_world SYSTEM INTERFACE "${OpenCV_INCLUDE_DIR}")
target_link_libraries(opencv_world INTERFACE "${OpenCV_CORE}" "${OpenCV_IMGPROC}" "${OpenCV_IMGCODECS}")
set(OpenCV_FOUND TRUE)
set(OpenCV_DIR "${CMAKE_CURRENT_SOURCE_DIR}/opencv")
function(slic3r_remap_configs targets from_Cfg to_Cfg)
 foreach(target IN LISTS targets)
  set_property(TARGET ${target} PROPERTY MAP_IMPORTED_CONFIG_${from_Cfg} ${to_Cfg})
 endforeach()
endfunction()
function(encoding_check target)
endfunction()
include_directories("${ORCA_SOURCE_DIR}/src" "${ORCA_SOURCE_DIR}/deps_src" "${CMAKE_BINARY_DIR}/orca")
foreach(dep agg ankerl earcut fast_float nanosvg nlohmann stb_dxt admesh clipper2 expat glu-libtess libigl libnest2d mcut miniz qhull qoi semver)
 add_subdirectory("${ORCA_SOURCE_DIR}/deps_src/${dep}" "deps/${dep}")
endforeach()
add_subdirectory("${ORCA_SOURCE_DIR}/src/libslic3r" "orca/libslic3r")
add_executable(orca-fltk-worker worker.cpp nanosvg.cpp ../src/profiles.cpp)
target_include_directories(orca-fltk-worker PRIVATE ../src)
target_link_libraries(orca-fltk-worker PRIVATE libslic3r libcurl)
target_link_directories(orca-fltk-worker PRIVATE "${CMAKE_PREFIX_PATH}/lib/x86_64-linux-gnu")
target_link_options(orca-fltk-worker PRIVATE -Wl,--disable-new-dtags)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../sources.json" SOURCE_LOCK)
string(JSON ORCA_PIN GET "${SOURCE_LOCK}" orca commit)
target_compile_definitions(orca-fltk-worker PRIVATE ORCA_PIN="${ORCA_PIN}")
set_target_properties(orca-fltk-worker PROPERTIES BUILD_RPATH "${CMAKE_PREFIX_PATH}/lib/x86_64-linux-gnu;${CMAKE_PREFIX_PATH}/lib")
install(TARGETS orca-fltk-worker RUNTIME DESTINATION libexec/orca-fltk COMPONENT OrcaRuntime)
